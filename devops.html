1. DevOps дегеніміз не?

DevOps — әзірлеушілер (Dev) мен пайдалану бөлімінің (Ops) бірлесіп жұмыс істеу мәдениеті және оны қолдайтын құралдар. Мақсаты — өзгерістерді жиі, жылдам және қауіпсіз шығару. DevOps бір бағдарлама емес: ол мәдениет, процесс және құралдар тізбегі.

Culture

ортақ жауапкершілік: «менің кодым» емес, «біздің сервис»

Automation

қайталанатын әрекеттің бәрі — скрипт пен конвейер

Lean

кішкентай өзгерістер, жиі шығару

Measurement · Sharing

метрикамен өлшеу, білімді бөлісу (CALMS)

5. CI/CD және DORA метрикалары

CI — әр өзгерісті автоматты тексеру (lint, тест, жинау). Continuous Delivery — кез келген сәтте шығаруға дайын артефакт. Continuous Deployment — тексеруден өткен өзгеріс өзі орналасады. Конвейер fail fast: бір кезең құласа, қалғандары орындалмайды. Артефакт бір рет жиналып, барлық ортаға сол бір файл жіберіледі.

# app.py — «Электронды журнал»: DevOps құралдары енгізілетін нақты жоба
#   python3 app.py          — http://127.0.0.1:8080
#   GET /health, GET /api/grades, POST /api/grades {"name": "...", "grade": 5}, GET /metrics
import json
import os
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

VERSION = open("VERSION").read().strip() if os.path.exists("VERSION") else "0.0.0-dev"
ENV = os.environ.get("APP_ENV", "dev")
PORT = int(os.environ.get("PORT", "8080"))
DATA = os.environ.get("DATA_FILE", "data/grades.json")
DEBUG = os.environ.get("DEBUG", "false") == "true"
STATS = {"requests": 0, "errors": 0, "started": time.time()}
DEFAULT = {"Айгерім": [5, 4, 5], "Нұрлан": [3, 4], "Дәулет": [2, 5, 4]}


def load():
    if not os.path.exists(DATA):
        return {k: list(v) for k, v in DEFAULT.items()}
    with open(DATA, encoding="utf-8") as f:
        return json.load(f)


def save(data):
    os.makedirs(os.path.dirname(DATA) or ".", exist_ok=True)
    with open(DATA, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=1)


def average(grades):
    return round(sum(grades) / len(grades), 2) if grades else 0.0


def validate(grade):
    return isinstance(grade, int) and not isinstance(grade, bool) and 1 <= grade <= 5


class Handler(BaseHTTPRequestHandler):
    def reply(self, code, data, ctype="application/json; charset=utf-8"):
        body = data.encode("utf-8") if isinstance(data, str) else json.dumps(data, ensure_ascii=False).encode("utf-8")
        STATS["requests"] += 1
        STATS["errors"] += code >= 500
        self.send_response(code)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path == "/health":
            return self.reply(200, {"status": "ok", "version": VERSION, "env": ENV})
        if self.path == "/api/grades":
            return self.reply(200, {n: {"grades": g, "avg": average(g)} for n, g in load().items()})
        if self.path == "/metrics":                         # Prometheus форматы
            return self.reply(200, "app_requests_total %d\napp_errors_total %d\napp_uptime_seconds %d\n"
                              'app_info{version="%s",env="%s"} 1\n' % (STATS["requests"], STATS["errors"],
                                                                       time.time() - STATS["started"], VERSION, ENV),
                              "text/plain; charset=utf-8")
        self.reply(404, {"error": "жол табылмады"})

    def do_POST(self):
        if self.path != "/api/grades":
            return self.reply(404, {"error": "жол табылмады"})
        try:
            req = json.loads(self.rfile.read(int(self.headers.get("Content-Length") or 0)))
        except ValueError:
            return self.reply(400, {"error": "JSON қате"})
        if not validate(req.get("grade")) or not req.get("name"):
            return self.reply(400, {"error": "баға 1–5 аралығындағы бүтін сан, name міндетті"})
        data = load()
        data.setdefault(req["name"], []).append(req["grade"])
        save(data)
        self.reply(201, {"name": req["name"], "avg": average(data[req["name"]])})

    def log_message(self, fmt, *args):
        if DEBUG:
            print("[%s] %s" % (ENV, fmt % args))


if __name__ == "__main__":
    print("Электронды журнал %s, орта: %s, http://127.0.0.1:%d" % (VERSION, ENV, PORT))
    ThreadingHTTPServer(("0.0.0.0", PORT), Handler).serve_forever()

# test_app.py — автоматты тесттер: python3 -m unittest -v
import os
import tempfile
import unittest

import app


class TestJournal(unittest.TestCase):
    def test_average(self):
        self.assertEqual(app.average([5, 4, 5]), 4.67)
        self.assertEqual(app.average([]), 0.0)

    def test_validate(self):
        self.assertTrue(app.validate(5))
        self.assertFalse(app.validate(6))
        self.assertFalse(app.validate(0))
        self.assertFalse(app.validate("5"))
        self.assertFalse(app.validate(True))

    def test_save_and_load(self):
        with tempfile.TemporaryDirectory() as d:
            app.DATA = os.path.join(d, "g.json")
            self.assertIn("Айгерім", app.load())          # файл жоқ — әдепкі деректер
            app.save({"Тест": [5]})
            self.assertEqual(app.load(), {"Тест": [5]})


if __name__ == "__main__":
    unittest.main()

# tasks.py — жобаның бірыңғай командалары (Makefile-дың барлық ОЖ-де жұмыс істейтін баламасы)
#   python3 tasks.py doctor   — компьютер жоба стандартына сай ма?
#   python3 tasks.py lint     — код стилі және синтаксис
#   python3 tasks.py test     — автоматты тесттер
#   python3 tasks.py build    — dist/journal-<нұсқа>.zip артефактісі
#   python3 tasks.py run [dev|test|prod] — .env.<орта> баптауымен іске қосу
import glob
import hashlib
import json
import os
import platform
import py_compile
import shutil
import subprocess
import sys
import zipfile

STD = json.load(open("devops.json", encoding="utf-8"))


def read_env(name):
    env = {}
    if os.path.exists(".env." + name):
        for line in open(".env." + name, encoding="utf-8"):
            line = line.strip()
            if line and not line.startswith("#") and "=" in line:
                k, v = line.split("=", 1)
                env[k.strip()] = v.strip()
    return env


def doctor():
    ok = True
    need = tuple(int(x) for x in STD["python"].split("."))
    have = sys.version_info[:2]
    print("%s Python %d.%d (стандарт: %s+)" % ("✓" if have >= need else "✗", *have, STD["python"]))
    ok &= have >= need
    for tool, level in STD["tools"].items():
        found = shutil.which(tool)
        print("%s %-7s %s" % ("✓" if found else ("✗" if level == "required" else "–"), tool,
                              found or ("ЖОҚ (міндетті)" if level == "required" else "жоқ (міндетті емес)")))
        ok &= bool(found) or level != "required"
    for e in STD["envs"]:
        missing = [v for v in STD["env_vars"] if v not in read_env(e)]
        print("%s .env.%s %s" % ("✗" if missing else "✓", e, "жетіспейді: " + ", ".join(missing) if missing else ""))
        ok &= not missing
    print("ОЖ: %s %s → %s" % (platform.system(), platform.release(), "ДАЙЫН" if ok else "СТАНДАРТҚА САЙ ЕМЕС"))
    return ok


def lint():
    problems = 0
    for path in sorted(glob.glob("*.py")):
        try:
            py_compile.compile(path, doraise=True)
        except py_compile.PyCompileError as e:
            print("%s: синтаксис қатесі: %s" % (path, e.msg))
            problems += 1
        for n, line in enumerate(open(path, encoding="utf-8"), 1):
            line = line.rstrip("\n")
            for bad, text in (("\t" in line, "TAB таңбасы"), (line != line.rstrip(), "жол соңындағы бос орын"),
                              (len(line) > STD["max_line"], "жол ұзын: %d > %d" % (len(line), STD["max_line"]))):
                if bad:
                    print("%s:%d: %s" % (path, n, text))
                    problems += 1
    print("lint: %d мәселе" % problems)
    return problems == 0


def test():
    return subprocess.run([sys.executable, "-m", "unittest", "-q"]).returncode == 0


def build():
    version = open("VERSION").read().strip()
    os.makedirs("dist", exist_ok=True)
    path = "dist/%s-%s.zip" % (STD["project"], version)
    with zipfile.ZipFile(path, "w", zipfile.ZIP_DEFLATED) as z:
        for f in STD["artifact"]:
            z.write(f)
    digest = hashlib.sha256(open(path, "rb").read()).hexdigest()
    open(path + ".sha256", "w").write(digest + "\n")
    print("артефакт: %s (%d байт), sha256: %s…" % (path, os.path.getsize(path), digest[:16]))
    return True


def run(name="dev"):
    env = dict(os.environ, APP_ENV=name, **read_env(name))
    return subprocess.run([sys.executable, "app.py"], env=env).returncode == 0


TASKS = {"doctor": doctor, "lint": lint, "test": test, "build": build, "run": run}
if __name__ == "__main__":
    if len(sys.argv) < 2 or sys.argv[1] not in TASKS:
        sys.exit("Қолданылуы: python3 tasks.py " + "|".join(TASKS))
    sys.exit(0 if TASKS[sys.argv[1]](*sys.argv[2:]) else 1)
  1. python3 tasks.py doctor → Python, git, docker, curl және үш .env файлының тексерісі, соңында «ДАЙЫН». Көршіңіздің компьютерімен салыстырыңыз — айырмашылық бар ма?
  2. python3 tasks.py lint → «lint: 0 мәселе». app.py-ге әдейі TAB немесе ұзын жол қосып, қайта тексеріңіз → файл:жол: мәселе. Түзетіңіз.
  3. python3 tasks.py test → OK. python3 tasks.py run test → журнал 8091 портында, орта test: curl http://127.0.0.1:8091/health.
  4. Аудит → ұпай өсті. Түсіндіріңіз: неге командаларды README-ге жазып қоймай, tasks.py-ге енгіздік? .editorconfig қандай мәселені шешеді (TAB/бос орын, CRLF/LF)?

import sys

errors = []

print("%-12s" % "кілт" + "".join("%-22s" % e for e in envs))

    row = []
    for e, vals in envs.items():
        if k not in vals:
            row.append("✗ ЖОҚ")
            errors.append("%s: .env.%s файлында жоқ" % (k, e))
        else:
            row.append(vals[k][:20])

    errors.append("prod: DEBUG=false болуы керек (қазір %s)" % prod.get("DEBUG"))

# pipeline.py — жергілікті CI/CD конвейері және DORA метрикалары
#   python3 pipeline.py            — lint → test → build → deploy(test) → smoke
#   python3 pipeline.py --report   — DORA метрикалары (metrics.jsonl бойынша)
import json
import os
import shutil
import subprocess
import sys
import time
import urllib.request
import zipfile

from tasks import STD, read_env


def sh(*args):
    return subprocess.run([sys.executable, "tasks.py", *args]).returncode == 0


def deploy():
    """Артефактіні deploy/test қалтасына ашып, сол жерден іске қосып, /health-ті тексеру (smoke test)."""
    version = open("VERSION").read().strip()
    target = "deploy/test"
    shutil.rmtree(target, ignore_errors=True)
    zipfile.ZipFile("dist/%s-%s.zip" % (STD["project"], version)).extractall(target)
    env = dict(os.environ, **read_env("test"))
    env["DATA_FILE"] = os.path.abspath(os.path.join(target, "data", "grades.json"))
    proc = subprocess.Popen([sys.executable, "app.py"], cwd=target, env=env, stdout=subprocess.DEVNULL)
    try:
        for _ in range(20):
            time.sleep(0.25)
            try:
                h = json.load(urllib.request.urlopen("http://127.0.0.1:%s/health" % env["PORT"], timeout=1))
                print("smoke: /health → %s, нұсқа %s, орта %s" % (h["status"], h["version"], h["env"]))
                return h["version"] == version and h["env"] == "test"
            except OSError:
                continue
        print("smoke: /health жауап бермеді")
        return False
    finally:
        proc.terminate()


STAGES = [("lint", lambda: sh("lint")), ("test", lambda: sh("test")), ("build", lambda: sh("build")),
          ("deploy", deploy)]


def commit():
    r = subprocess.run(["git", "rev-parse", "--short", "HEAD"], capture_output=True, text=True)
    return r.stdout.strip() or "local"


def run():
    start, failed = time.time(), None
    for name, step in STAGES:
        t = time.time()
        print("\n▶ %s" % name)
        ok = step()
        print("%s %s (%.1f с)" % ("✓" if ok else "✗", name, time.time() - t))
        if not ok:
            failed = name
            break                                        # fail fast: келесі кезеңдер орындалмайды
    rec = {"ts": int(time.time()), "commit": commit(), "version": open("VERSION").read().strip(),
           "result": "fail" if failed else "success", "failed_stage": failed, "seconds": round(time.time() - start, 1)}
    with open("metrics.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps(rec) + "\n")
    print("\nКОНВЕЙЕР: %s" % ("СӘТТІ" if not failed else "ҚҰЛАДЫ (%s кезеңі)" % failed))
    return not failed


def report():
    runs = [json.loads(line) for line in open("metrics.jsonl", encoding="utf-8")]
    ok = [r for r in runs if r["result"] == "success"]
    fails = [r for r in runs if r["result"] == "fail"]
    restore, broken = [], None
    for r in runs:                                       # MTTR: құлағаннан келесі сәттіге дейін
        if r["result"] == "fail" and broken is None:
            broken = r["ts"]
        elif r["result"] == "success" and broken is not None:
            restore.append(r["ts"] - broken)
            broken = None
    days = max(1, (runs[-1]["ts"] - runs[0]["ts"]) / 86400)
    print("Іске қосулар: %d (сәтті %d, сәтсіз %d)" % (len(runs), len(ok), len(fails)))
    print("Орналастыру жиілігі: %.1f / күн" % (len(ok) / days))
    print("Өзгерістердің сәтсіздік үлесі (CFR): %.0f%%" % (100 * len(fails) / len(runs)))
    print("Конвейердің орташа уақыты (lead time-қа жуық): %.1f с" % (sum(r["seconds"] for r in runs) / len(runs)))
    print("Қалпына келтіру уақыты (MTTR): %s" % ("%.0f с" % (sum(restore) / len(restore)) if restore else "—"))


if __name__ == "__main__":
    if "--report" in sys.argv:
        report()
    else:
        sys.exit(0 if run() else 1)

  4. GitHub (қосымша): жобаны GitHub-қа жүктеп (git remote add origin ..., git push), Actions қойындысында CI-дің өтуін көріңіз.
  5. Түсіндіріңіз: CI мен CD айырмашылығы? Неге артефакт бір рет жиналып, барлық ортаға сол бір файл жіберіледі? Сіздің CFR мен MTTR мәндеріңіз нені көрсетеді?

Есепке: pipeline.py, ci.yml, сәтті және құлаған конвейердің скриншоттары, DORA есебі, сұрақтарға жауап.

6-тапсырма. Енгізудің кешенді жоспары, жеке нұсқа (30 мин, 20 ұпай)

Енгізу жоспары аудиттің бастапқы нәтижесінен құрылады: әр кемшілік — жұмыс, жұмыстар фазаларға бөлінеді, әр фазаның жауаптысы, мерзімі, мақсатты метрикалары (KPI), тәуекелдері және жауапкершілік кестесі (RACI) болады. Біз оны қолмен емес, деректен жасаймыз — жоба өзгерсе, жоспар да қайта жасалады.

Файл: plan_builder.py

# plan_builder.py — аудит нәтижесі мен команда мүмкіндігінен DevOps енгізудің кешенді жоспарын жасау
#   python3 devops_audit.py && python3 plan_builder.py   → PLAN.md
# audit_start.json — бастапқы жағдай (жоспардың жұмыстары), audit.json — қазіргі жағдай (не орындалды)
import datetime as dt
import json

audit = json.load(open("audit_start.json", encoding="utf-8"))
done = {r["check"] for r in json.load(open("audit.json", encoding="utf-8"))["results"] if r["ok"]}
inp = json.load(open("plan_input.json", encoding="utf-8"))
start = dt.date.fromisoformat(inp["start"])
roles = {m["role"]: m["name"] for m in inp["team"]}

PHASES = [  # (фаза, кезеңдер, жауапты рөл, апта саны)
    ("0. Бағалау және келісу", ["plan"], "lead", 1),
    ("1. Негіз: код пен орта стандарты", ["code", "build"], "dev", 2),
    ("2. Сапа: тесттер және CI", ["test", "ci"], "qa", 2),
    ("3. Шығару және орналастыру", ["release", "deploy"], "ops", 2),
    ("4. Пайдалану және мониторинг", ["operate", "monitor"], "ops", 1),
]
todo = [r for r in audit["results"] if not r["ok"]]
lines = ["# %s: DevOps құралдарын енгізу жоспары" % inp["project"], "",
         "Бастапқы жағдай: аудит ұпайы **%d/100**, деңгей: %s. Мерзімі: %s бастап %d апта. Команда: %s." % (
             audit["score"], audit["level"], start.strftime("%d.%m.%Y"), sum(p[3] for p in PHASES),
             ", ".join("%s (%s)" % (m["name"], m["role"]) for m in inp["team"])), "",
         "## Мақсат (KPI)", "", "| Метрика | Қазір | Мақсат |", "|---|---|---|"]
lines += ["| %s | %s | %s |" % (k["metric"], k["now"], k["target"]) for k in inp["kpi"]]
lines += ["", "## Фазалар мен жұмыстар", ""]
week, gantt = 0, []
for name, stages, role, weeks in PHASES:
    items = [r for r in todo if r["stage"] in stages]
    begin, end = start + dt.timedelta(weeks=week), start + dt.timedelta(weeks=week + weeks) - dt.timedelta(days=1)
    lines += ["### %s (%s – %s), жауапты: %s" % (name, begin.strftime("%d.%m"), end.strftime("%d.%m"),
                                                 roles.get(role, role)), ""]
    lines += ["- [%s] %s — %s" % ("x" if r["check"] in done else " ", r["check"], r["fix"]) for r in items] or [
        "- [x] бастапқы аудит бойынша бұл фазада кемшілік жоқ — сапасын тексеру"]
    lines += ["- [ ] Фаза нәтижесін демо және ретроспектива", ""]
    gantt.append("%-38s %s%s" % (name[:38], "  " * week, "██" * weeks))
    week += weeks
header = "%-38s %s" % ("апта:", "".join("%-2d" % (i + 1) for i in range(week)))
lines += ["## Кесте (Гант)", "", "```", header] + gantt + ["```", ""]
lines += ["## Тәуекелдер", "", "| Тәуекел | Ықтималдығы | Шара |", "|---|---|---|"]
lines += ["| %s | %s | %s |" % tuple(r) for r in inp["risks"]]
lines += ["", "## Жауапкершілік (RACI)", "", "| Жұмыс | " + " | ".join(roles.values()) + " |",
          "|---|" + "---|" * len(roles)]
raci = {"Орта стандарты": "dev", "CI конвейері": "qa", "Орналастыру": "ops", "Жоспар және бақылау": "lead"}
for work, owner in raci.items():
    cells = ["R/A" if r == owner else ("C" if r == "lead" else "I") for r in roles]
    lines.append("| %s | %s |" % (work, " | ".join(cells)))
finished = sum(r["check"] in done for r in todo)
lines += ["", "**Орындалуы:** %d / %d жұмыс (%d%%)." % (finished, len(todo), 100 * finished // max(1, len(todo))),
          "", "_Жоспар devops_audit.py нәтижесінен автоматты түрде жасалды._"]
open("PLAN.md", "w", encoding="utf-8").write("\n".join(lines) + "\n")
print("PLAN.md жасалды: %d фаза, %d жұмыс, %d апта, орындалғаны: %d" % (len(PHASES), len(todo), week, finished))
  1. python3 devops_audit.py, содан кейін python3 plan_builder.py → «PLAN.md жасалды: 5 фаза, N жұмыс, 8 апта, орындалғаны: N».
  2. PLAN.md-ді VS Code-та ашыңыз (Cmd+Shift+V — markdown көрінісі): KPI, фазалар, [x] орындалған жұмыстар, Гант кестесі, тәуекелдер, RACI.
  3. plan_input.json-ды өз командаңызға бейімдеңіз: аты-жөндер, бастау күні, 1 жаңа тәуекел, KPI мақсаттары. Жоспарды қайта жасаңыз.
  4. Аудиттің соңғы ұпайы → 100/100. Бастапқы және соңғы ұпайды салыстырыңыз.
  5. Нұсқаңыз бойынша тапсырманы орындаңыз (өз жұмысыңыз).

Бақылау сұрақтары

  1. DevOps дегеніміз не және ол қандай мәселелерді шешеді?
  2. DevOps құралдар тізбегінің кезеңдерін және әр кезеңнің құралын атаңыз.
  3. «Менің компьютерімде жұмыс істейді» мәселесі неден туындайды және оны қалай шешеміз?
  4. Орталарды бір стандартқа келтіру нені білдіреді? Ортаның дрейфі дегеніміз не?
  5. Неге бір Docker образы барлық ортада қолданылады, ал айырмашылық орта айнымалыларында болады?
  6. CI, Continuous Delivery және Continuous Deployment айырмашылығы?
  7. Fail fast принципі және артефакт дегеніміз не?
  8. DORA метрикаларының төртеуін атап, мағынасын түсіндіріңіз.
  9. Енгізу жоспарында не болуы керек (фазалар, жауаптылар, мерзім, KPI, тәуекелдер, RACI)?
  10. Неге DevOps-ты бірден барлық жерге емес, фазалармен енгізген дұрыс?
